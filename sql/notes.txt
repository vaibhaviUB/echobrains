BETWEEN OPERATOR
----------------

BETWEEN is an special operator used to filter  records where condition lies on specific range. we will use between after where
syntax- select * from table where colname between start and end;


1) waqtd emp whos salary is > 10k and < 35k but salary is not equal to 25000

 	select * from employee where salary >= 10000 and salary <= 35000 and salary != 25000;	(using and op)
	
	select * from employee where salary between 10000 and 35000; 	(using between)

2) waqtd emp name and salary and joining date who joined in between 2020 and 2025

	select name, salary, joining_date from employee where joining_date between '2020-01-01' and '2025-12-31';

3) list the emp who not joined in 2025 and salary is > 35k and working in sales dept or location = Chennai 
	
	SELECT * FROM employee WHERE not joining_date between '2025-01-01' and '2025-12-31' AND salary > 35000 AND department = 'Sales' OR city = 'Chennai';
	
	SELECT * FROM employee WHERE not joining_date < '2025-01-01' or joining_date > '2025-12-31' AND salary > 35000 AND department = 'Sales' OR city = 'Chennai';

4) filter emp whos salary not between 32000 to 45000

	select * from employee where salary not between 32000 and 45000;

PATTERN MATCHING - LIKE OPERATOR
--------------------------------
USED TO filter records based on the pattern matching 
% -- for one or more character
_ -- for exactly one character / single character

syntax-- select * from tablename where colname like '_or%';

 
1)filter employee whos name starts with character A 

	select * from employee where name like 'a%';

2)filter employee whos name have 4 character

3) waqtd emp who is having h in their name

ORDER BY 
--------

used to sort the result (asc or des)
order by comes after where clause(if we use where in  query)
order by will work on numeric, strings, dates
we can sort the result with multiple cols
asc= low to high
desc= high to low

syntx -  
select colname from tablename order by colname  asc/desc;

1) display emp based on salary asc
 
	select * from employee order by salary asc;

2) display emps based on salary higher to lower

	select * from employee order by salary desc;

3) display emp based on salary low to high , and exp

	select * from employee order by salary,exp asc; (always preference is for 1st col)

4) sort the name, annualsalary emps based on annual salary  high to low

	select name,salary*12 as annual_Salary from employee order by desc;  (using alias name)

	
5)waqtd emp who is working in sales dept sort them based on their exp

	select * from employee where department='Sales' order by exp;

6) waqtd emps name, dept, exp, from employees whose total salary is greater than 4lakhs and sort them based on total salary ;

	select name, department, exp from employee where ((salary + bonus)*12) > 400000 order by (salary+bonus);


LIMIT
-----
to get limited rows irrespective of any condition
used to select specific num of rows from the table

1) select only 2 emp rows

	select * from employee limit 2;

2) display top paid employee from the table (only 1)

	select * from employee order by salary desc limit 1;


OFFSET
------
OFFSET tells SQL how many records to skip before returning the result.
used in pagination

1) display emp  who salary is second highest (2nd hightest paid emp)

	select * from employee order by salary desc limit 1 offset 1;

2) suppose if we have no of employee and we want to display only 3 emps in each page of frontend , so this is how we do it 
	select * from employee order by salary desc limit 3 offset 0;
	select * from employee order by salary desc limit 3 offset 3;
	select * from employee order by salary desc limit 3 offset 6;
	
3) waqtd top paid emp in finance dep
