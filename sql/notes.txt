BETWEEN OPERATOR
----------------

BETWEEN is an special operator used to filter  records where condition lies on specific range. we will use between after where
syntax- select * from table where colname between start and end;


1) waqtd emp whos salary is > 10k and < 35k but salary is not equal to 25000

 	select * from employee where salary >= 10000 and salary <= 35000 and salary != 25000;	(using and op)
	
	select * from employee where salary between 10000 and 35000; 	(using between)

2) waqtd emp name and salary and joining date who joined in between 2020 and 2025

	select name, salary, joining_date from employee where joining_date between '2020-01-01' and '2025-12-31';

3) list the emp who not joined in 2025 and salary is > 35k and working in sales dept or location = Chennai 
	
	SELECT * FROM employee WHERE not joining_date between '2025-01-01' and '2025-12-31' AND salary > 35000 AND department = 'Sales' OR city = 'Chennai';
	
	SELECT * FROM employee WHERE not joining_date < '2025-01-01' or joining_date > '2025-12-31' AND salary > 35000 AND department = 'Sales' OR city = 'Chennai';

4) filter emp whos salary not between 32000 to 45000

	select * from employee where salary not between 32000 and 45000;

PATTERN MATCHING - LIKE OPERATOR
--------------------------------
USED TO filter records based on the pattern matching 
% -- for one or more character
_ -- for exactly one character / single character

syntax-- select * from tablename where colname like '_or%';

 
1)filter employee whos name starts with character A 

	select * from employee where name like 'a%';

2)filter employee whos name have 4 character 

	select * from employee where name like '____'; (use 4 underscore)

3) waqtd emp who is having h in their name 

	select * from employee where name like '%h%';

4)  waqtd emp whos 3rd letter contains h 

	select * from employee where name like '__h%';
